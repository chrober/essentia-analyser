cmake_minimum_required(VERSION 3.16)
project(essentia_ipc_fixture LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(essentia-pe-fixture-dependency SHARED ../pe_fixture_dependency.cpp)
add_executable(essentia-protocol-fixture ../protocol_fixture.cpp)
target_compile_definitions(essentia-protocol-fixture PRIVATE ESSENTIA_PE_FIXTURE_DEP)
target_link_libraries(essentia-protocol-fixture PRIVATE essentia-pe-fixture-dependency)
if(MSVC)
    target_compile_options(essentia-protocol-fixture PRIVATE /W4)
else()
    target_compile_options(essentia-protocol-fixture PRIVATE -Wall -Wextra)
endif()
